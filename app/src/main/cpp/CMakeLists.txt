cmake_minimum_required(VERSION 3.22.1)
project(madar_audio)
add_library(sherpa SHARED IMPORTED)
set_target_properties(sherpa PROPERTIES IMPORTED_LOCATION "${CMAKE_CURRENT_SOURCE_DIR}/../jniLibs/${ANDROID_ABI}/libsherpa-onnx-c-api.so")
add_library(mp3lame STATIC IMPORTED)
set_target_properties(mp3lame PROPERTIES IMPORTED_LOCATION "${CMAKE_CURRENT_SOURCE_DIR}/../../../vendor/${ANDROID_ABI}/libmp3lame.a")
add_library(madar_audio SHARED separation.cpp)
target_include_directories(madar_audio PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/../../../vendor")
target_link_libraries(madar_audio sherpa mp3lame log m)
target_link_options(madar_audio PRIVATE "-Wl,-z,max-page-size=16384")
foreach(component avformat avcodec avutil)
 add_library(${component} STATIC IMPORTED)
 set_target_properties(${component} PROPERTIES IMPORTED_LOCATION "${CMAKE_CURRENT_SOURCE_DIR}/../../../vendor/ffmpeg-${ANDROID_ABI}/lib/lib${component}.a")
endforeach()
target_include_directories(madar_audio PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/../../../vendor/ffmpeg-${ANDROID_ABI}/include")
target_link_libraries(madar_audio avformat avcodec avutil z)
